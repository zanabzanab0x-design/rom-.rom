const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const path = require('path'); const app = express(); const server = http.createServer(app); const io = new Server(server); // إعدادات المالك الحصرية const OWNER_CODE = "12233"; const OWNER_NAME = "كرامه"; let isChatLocked = false; app.use(express.static(path.join(__dirname, 'public'))); io.on('connection', (socket) => { // تعريف اسم افتراضي للمستخدم العادي socket.username = "مستخدم_" + socket.id.substring(0, 4); // استقبال الرسائل socket.on('sendMessage', (data) => { // منع المستخدم العادي إذا كان الشات مغلقاً if (isChatLocked && !socket.isOwner) { socket.emit('adminAction', { type: 'system', text: '⚠️ الشات مغلق حالياً من قِبل المالك كرامه!' }); return; } // إرسال الرسالة حسب الرتبة if (socket.isOwner) { io.emit('newMessage', { sender: `👑 المالك ${OWNER_NAME}`, text: data.text, type: 'owner-sent' }); } else { // إرسال للمرسل بشكل منفصل وللبقية بشكل منفصل للتحكم بالتصميم socket.emit('newMessage', { sender: "أنت", text: data.text, type: 'sent' }); socket.broadcast.emit('newMessage', { sender: socket.username, text: data.text, type: 'received' }); } }); // التحقق من كود المالك socket.on('checkAdmin', (code) => { if (code === OWNER_CODE) { socket.isOwner = true; socket.username = OWNER_NAME; socket.emit('adminVerified', { success: true, ownerName: OWNER_NAME }); io.emit('newMessage', { sender: "النظام", text: `🔓 سجل المالك (${OWNER_NAME}) دخوله للتحكم بالغرفة.`, type: 'admin-msg' }); } else { socket.emit('adminVerified', { success: false }); } }); // لوحة التحكم بكامل الصلاحيات socket.on('adminCommand', (action) => { if (!socket.isOwner) return; // حماية السيرفر if (action.type === 'clear') { io.emit('adminAction', { type: 'clear' }); io.emit('newMessage', { sender: "النظام", text: `🧹 قام المالك ${OWNER_NAME} بمسح سجل المحادثات للجميع.`, type: 'admin-msg' }); } else if (action.type === 'lock') { isChatLocked = !isChatLocked; const statusText = isChatLocked ? `🔒 قام المالك ${OWNER_NAME} بقفل المحادثة.` : `🔓 قام المالك ${OWNER_NAME} بفتح المحادثة للجميع.`; io.emit('newMessage', { sender: "النظام", text: statusText, type: 'admin-msg' }); } else if (action.type === 'alert') { io.emit('newMessage', { sender: `📢 تنبيه من المالك ${OWNER_NAME}`, text: action.text, type: 'admin-msg' }); } }); }); const PORT = 3000; server.listen(PORT, () => { console.log(`🚀 السيرفر يعمل بنجاح على: http://localhost:${PORT}`); });
